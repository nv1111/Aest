    66→    66→function initials(name: string) {
    67→    67→  return name
    68→    68→    .split(/\s+/)
    69→    69→    .map((p) => p[0])
    70→    70→    .slice(0, 2)
    71→    71→    .join("")
    72→    72→    .toUpperCase();
    73→    73→}

    75→    75→function mmss(seconds: number): string {
    76→    76→  const m = Math.floor(seconds / 60);
    77→    77→  const s = seconds % 60;
    78→    78→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
    79→    79→}

   137→   137→        (old) =>
   138→   138→          old

   140→   140→                ...old,
   141→   141→                messages: old.messages.map((x) =>
   142→   142→                  x.senderRole === "user" && !x.readAt ? { ...x, readAt } : x

   235→   235→        setPending((p) => p.filter((x) => x.localId !== m.localId));
   236→   236→        appendMessage(message);
   237→   237→      } catch (err) {
   238→   238→        setPending((p) => p.map((x) => (x.localId === m.localId ? { ...x, status: "failed" } : x)));
   239→   239→        if (err instanceof ApiError && err.code === "insufficient_balance") {
   240→   240→          setInsufficientOpen(true);
   241→   241→          void qc.invalidateQueries({ queryKey: ["me"] });
   242→   242→        } else {
   243→   243→          toast.error(errorMessage(err));
   244→   244→        }
   245→   245→      }
   246→   246→    },
   247→   247→    [id, appendMessage, qc]

   249→   249→

   260→   260→    } catch (err) {
   261→   261→      toast.error(errorMessage(err));
   262→   262→      setEnding(false);
   263→   263→    }
   264→   264→  }, [id, isActive, ending, qc, replace]);
   265→   265→
   266→   266→  const handleBack = () => {
   267→   267→    if (isActive) {
   268→   268→      setBackOpen(true);
   269→   269→      return;
   270→   270→    }
   271→   271→    pop();
   272→   272→  };
   273→   273→
   274→   274→  // ------------------------------------------------------------ render
   275→   275→  if (!id) {
   276→   276→    return (
   277→   277→      <ScreenScaffold bare>
   278→   278→        <ErrorState title={t("common.errorGeneric")} />
   279→   279→      </ScreenScaffold>
   280→   280→    );
   281→   281→  }
   282→   282→

   286→   286→        <PageSkeleton variant="list" />
   287→   287→      </ScreenScaffold>

   289→   289→  }
   290→   290→

   292→   292→    return (
   293→   293→      <ScreenScaffold bare>
   294→   294→        <ErrorState title={t("astrologers.notFoundTitle")} onRetry={() => query.refetch()} />

   296→   296→    );
   297→   297→  }

   299→   299→  const a = consultation.astrologer;
   300→   300→  const ModeIcon = MODE_ICON[(consultation.mode as keyof typeof MODE_ICON) ?? "chat"] ?? MessageCircle;
   301→   301→
   302→   302→  const startedMs = consultation.startedAt ? new Date(consultation.startedAt).getTime() : null;
   303→   303→  const elapsedSec =

   332→   332→                  {initials(a.displayName)}
   333→   333→                </AvatarFallback>
   334→   334→              </Avatar>
   335→   335→              {a.onlineStatus === "online" ? (
   336→   336→                <span className="absolute -bottom-0.5 -right-0.5 h-2.5 w-2.5 rounded-full border-2 border-background bg-success" aria-label={t("common.online")} />
   337→   337→              ) : null}
   338→   338→            </div>
   339→   339→            <div className="min-w-0 flex-1">
   340→   340→              <div className="flex items-center gap-1.5">

   342→   342→                  {a.displayName}

   437→   437→      {/* ------------------------------------------------------------ input */}
   438→   438→      {isActive ? (
   439→   439→        <div className="shrink-0 border-t bg-background px-3 py-2.5">

   449→   449→              onChange={(e) => setDraft(e.target.value)}
