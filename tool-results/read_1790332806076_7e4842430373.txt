    76→interface PendingExchange {

    82→// ------------------------------------------------------- speech recognition
    83→// Minimal structural types — SpeechRecognition is not in the TS DOM lib and
    84→// availability varies (webkit prefix). Everything is narrowed to unknown.
    85→
    86→interface RecognitionResultItem {
    87→  transcript: string;
    88→}
    89→interface RecognitionResultEvent {
    90→  results: ArrayLike<ArrayLike<RecognitionResultItem>>;
    91→}
    92→interface RecognitionLike {
    93→  lang: string;
    94→  interimResults: boolean;
    95→  continuous: boolean;
    96→  onresult: ((event: RecognitionResultEvent) => void) | null;
    97→  onerror: (() => void) | null;
    98→  onend: (() => void) | null;
    99→  start: () => void;
   100→  stop: () => void;
   101→}
   102→type RecognitionCtor = new () => RecognitionLike;
   103→
   104→function getRecognitionCtor(): RecognitionCtor | null {
   105→  if (typeof window === "undefined") return null;
   106→  const w = window as unknown as {
   107→    SpeechRecognition?: RecognitionCtor;
   108→    webkitSpeechRecognition?: RecognitionCtor;
   109→  };
   110→  return w.SpeechRecognition ?? w.webkitSpeechRecognition ?? null;
   111→}
   112→
   113→function isAbortError(err: unknown): boolean {
   114→  return err instanceof Error && err.name === "AbortError";
   115→}

   117→// ------------------------------------------------------------- audio playback
   118→// One audio element for the whole screen — only one reading speaks at a time.

   120→let sharedAudio: HTMLAudioElement | null = null;
   121→function getSharedAudio(): HTMLAudioElement {
   122→  if (!sharedAudio) sharedAudio = new Audio();
   123→  return sharedAudio;
   124→}

   130→/**
   131→ * AskScreen — the "Ask your chart" conversational assistant.
   132→ * ChatGPT-style flow grounded in the user's pre-computed chart context.
   133→ */
   134→export default function AskScreen() {
   135→  const me = useMe();
   136→  const qc = useQueryClient();
   137→  const push = useAppStore((s) => s.push);
   138→  const openInTab = useAppStore((s) => s.openInTab);
   139→  const screen = useCurrentScreen();
   140→  const qParam = typeof screen.params?.q === "string" ? screen.params.q : undefined;
   141→
   142→  const [input, setInput] = useState("");
   143→  const [activeId, setActiveIdState] = useState<string | null>(() => {
   144→    if (typeof window === "undefined") return null;
   145→    try {
   146→      return window.sessionStorage.getItem(CONV_STORAGE_KEY);
   147→    } catch {
   148→      return null;
   149→    }
   150→  });
   151→  const [pending, setPending] = useState<PendingExchange | null>(null);
   152→  const [expanded, setExpanded] = useState<Record<string, boolean>>({});
   153→  const [feedbackSent, setFeedbackSent] = useState<Record<string, 1 | -1>>({});
   154→  const [historyOpen, setHistoryOpen] = useState(false);
   155→  const [listening, setListening] = useState(false);
   156→  const [speakState, setSpeakState] = useState<SpeakState>(null);
   157→
   158→  const pendingRef = useRef<PendingExchange | null>(null);
   159→  const abortRef = useRef<AbortController | null>(null);
   160→  const recognitionRef = useRef<RecognitionLike | null>(null);
   161→  const inputRef = useRef<HTMLTextAreaElement>(null);
   162→  const listRef = useRef<HTMLDivElement>(null);
   163→  const autoSentRef = useRef(false);
   164→  const speakTokenRef = useRef(0);
   165→
   166→  const hasProfile = !!me.data?.primaryProfile;
   167→
   168→  // ------------------------------------------------------------- data
   169→
   170→  const conversations = useQuery({
   171→    queryKey: ["ai-conversations"],
   172→    queryFn: () => aiService.conversations(),
   173→    enabled: !!me.data?.user,
   174→    staleTime: 30_000,
   175→  });
   176→
   177→  const messagesQuery = useQuery<AskMessageDTO[]>({
   178→    queryKey: ["ai-messages", activeId],
   179→    queryFn: async () => (await aiService.messages(activeId as string)).messages,
   180→    enabled: !!activeId,
   181→    staleTime: 30_000,
   182→    retry: false,
   183→  });
   184→
   185→  const messages = activeId ? (messagesQuery.data ?? []) : [];
   186→  const messagesLoading = !!activeId && messagesQuery.isLoading;
   187→  const messagesError =
   188→    !!activeId && messagesQuery.isError && !(messagesQuery.error instanceof ApiError && messagesQuery.error.code === "not_found");
   189→
   190→  const activeConversation: AiConversationSummaryDTO | undefined = conversations.data?.conversations.find(
   191→    (c) => c.id === activeId
   192→  );
   193→
   194→  // ------------------------------------------------------------- actions
   195→
   196→  const setActive = useCallback((id: string | null) => {
   197→    setActiveIdState(id);
   198→    try {
   199→      if (id) window.sessionStorage.setItem(CONV_STORAGE_KEY, id);
   200→      else window.sessionStorage.removeItem(CONV_STORAGE_KEY);
   201→    } catch {
   202→      /* storage unavailable — conversation still works in-memory */
   203→    }
   204→  }, []);
   205→
   206→  // A stored conversation that no longer exists (cleared) → soft reset.
   207→  useEffect(() => {
   208→    if (messagesQuery.error instanceof ApiError && messagesQuery.error.code === "not_found") {
   209→      setActive(null);
   210→    }
   211→  }, [messagesQuery.error, setActive]);
   212→
   213→  const send = useCallback(
   214→    async (text: string) => {
   215→      const question = text.trim().slice(0, MAX_MESSAGE_LENGTH);
   216→      if (!question || pendingRef.current) return;
   217→      setInput("");
   218→      const exchange: PendingExchange = { question, status: "thinking" };
   219→      pendingRef.current = exchange;
   220→      setPending(exchange);
   221→      const controller = new AbortController();
   222→      abortRef.current = controller;
   223→      try {
   224→        const result = await aiService.ask(question, activeId ?? undefined, controller.signal);
   225→        trackEvent("ai_question_asked");
   226→        qc.setQueryData<AskMessageDTO[]>(["ai-messages", result.conversationId], (old) =>
   227→          old ? [...old, ...result.messages] : [...result.messages]
   228→        );
   229→        setActive(result.conversationId);
   230→        pendingRef.current = null;
   231→        setPending(null);
   232→        void qc.invalidateQueries({ queryKey: ["ai-conversations"] });
   233→      } catch (err) {
   234→        const failure: PendingExchange = {
   235→          question,
   236→          status: "failed",
   237→          error: isAbortError(err) ? t("ask.stopped") : errorMessage(err),
   238→        };
   239→        pendingRef.current = failure;
   240→        setPending(failure);
   241→      } finally {
   242→        abortRef.current = null;
   243→      }
   244→    },
   245→    [activeId, qc, setActive]
   246→  );
   247→
   248→  const retryPending = useCallback(() => {
   249→    const current = pendingRef.current;
   250→    if (!current || current.status !== "failed") return;
   251→    pendingRef.current = null;
   252→    setPending(null);
   253→    void send(current.question);
   254→  }, [send]);
   255→
   256→  const stopThinking = useCallback(() => {
   257→    abortRef.current?.abort();
   258→  }, []);
   259→
   260→  // Home can navigate here with a prefilled question (params.q).
   261→  useEffect(() => {
   262→    if (autoSentRef.current) return;
   263→    if (me.isLoading) return;
   264→    const q = (qParam ?? "").trim();
   265→    if (!q) return;
   266→    autoSentRef.current = true;
   267→    if (hasProfile) {
   268→      void send(q);
   269→    } else {
   270→      setInput(q.slice(0, MAX_MESSAGE_LENGTH));
   271→    }
   272→  }, [me.isLoading, hasProfile, qParam, send]);
   273→
   274→  const toggleFactors = useCallback((id: string) => {
   275→    setExpanded((prev) => ({ ...prev, [id]: !prev[id] }));
   276→  }, []);
   277→
   278→  const sendFeedback = useCallback(async (messageId: string, value: 1 | -1) => {
   279→    setFeedbackSent((prev) => ({ ...prev, [messageId]: value }));
   280→    try {
   281→      await aiService.feedback(messageId, value);
   282→      toast(t("ask.thanksForFeedback"));
   283→    } catch (err) {
   284→      toast.error(errorMessage(err));
   285→    }
   286→  }, []);
   287→
   288→  const copyAnswer = useCallback(async (content: string) => {
   289→    try {
   290→      await navigator.clipboard.writeText(content);
   291→      toast(t("ask.copied"));
   292→    } catch {
   293→      toast.error(t("common.errorGeneric"));
   294→    }
   295→  }, []);
   296→
   297→  const toggleListen = useCallback(
   298→    async (messageId: string) => {
   299→      const audio = getSharedAudio();
   300→
   301→      // tap on the speaking/loading message → stop
   302→      if (speakState?.id === messageId) {
   303→        speakTokenRef.current += 1; // invalidate in-flight loads
   304→        audio.pause();
   305→        audio.removeAttribute("src"); // release the decoder
   306→        setSpeakState(null);
   307→        return;
   308→      }
   309→
   310→      // another message is speaking → switch
   311→      speakTokenRef.current += 1;
   312→      audio.pause();
   313→      const token = speakTokenRef.current;
   314→      setSpeakState({ id: messageId, status: "loading" });
   315→      try {
   316→        const url = await aiService.speak(messageId);
   317→        if (token !== speakTokenRef.current) return; // superseded meanwhile
   318→        audio.src = url;
   319→        audio.onended = () => setSpeakState((s) => (s?.id === messageId ? null : s));
   320→        audio.onerror = () => setSpeakState((s) => (s?.id === messageId ? null : s));
   321→        await audio.play();
   322→        if (token !== speakTokenRef.current) {
   323→          audio.pause();
   324→          return;
   325→        }
   326→        setSpeakState({ id: messageId, status: "playing" });
   327→        trackEvent("reading_listened");
   328→      } catch (err) {
   329→        if (token !== speakTokenRef.current) return;
   330→        setSpeakState(null);
   331→        if (err instanceof DOMException && (err.name === "NotAllowedError" || err.name === "AbortError")) return;
   332→        toast.error(errorMessage(err));
   333→      }
   334→    },
   335→    [speakState]
   336→  );
   337→
   338→  // stop the voice when leaving the screen
   339→  useEffect(() => {
   340→    return () => {
   341→      speakTokenRef.current += 1;
   342→      const audio = getSharedAudio();
   343→      audio.pause();
   344→      audio.onended = null;
   345→      audio.onerror = null;
   346→    };
   347→  }, []);
   348→
   349→  const shareAnswer = useCallback(async (content: string) => {
   350→    if (typeof navigator !== "undefined" && typeof navigator.share === "function") {
   351→      try {
   352→        await navigator.share({ title: t("common.appName"), text: content });
   353→      } catch {
   354→        /* user dismissed the share sheet */
   355→      }
   356→      return;
   357→    }
   358→    try {
   359→      await navigator.clipboard.writeText(content);
   360→      toast(t("ask.shareCopied"));
   361→    } catch {
   362→      toast.error(t("common.errorGeneric"));
   363→    }
   364→  }, []);
   365→
   366→  const loadConversation = useCallback(
   367→    (id: string) => {
   368→      setActive(id);
   369→      setExpanded({});
   370→      setHistoryOpen(false);
   371→    },
   372→    [setActive]
   373→  );
   374→
   375→  const startNew = useCallback(() => {
   376→    setActive(null);
   377→    setExpanded({});
   378→  }, [setActive]);
   379→
   380→  const clearAll = useCallback(async () => {
   381→    try {
   382→      await aiService.clearAll();
   383→      qc.removeQueries({ queryKey: ["ai-messages"] });
   384→      setActive(null);
   385→      setHistoryOpen(false);
   386→      void qc.invalidateQueries({ queryKey: ["ai-conversations"] });
   387→      toast(t("ask.clearedToast"));
   388→    } catch (err) {
   389→      toast.error(errorMessage(err));
   390→    }
   391→  }, [qc, setActive]);
   392→
   393→  // ------------------------------------------------------------- voice input
   394→
   395→  const toggleMic = useCallback(() => {
   396→    if (listening) {
   397→      recognitionRef.current?.stop();
   398→      return;
   399→    }
   400→    const Ctor = getRecognitionCtor();
   401→    if (!Ctor) {
   402→      toast(t("ask.voiceUnavailable"));
   403→      return;
   404→    }
   405→    const rec = new Ctor();
   406→    rec.lang = "en-IN";
   407→    rec.interimResults = true;
   408→    rec.continuous = false;
   409→    rec.onresult = (event) => {
   410→      let transcript = "";
   411→      const list = event.results;
   412→      for (let i = list.length - 1; i >= 0; i--) {
   413→        const alt = list[i]?.[0];
   414→        if (alt?.transcript) {
   415→          transcript = alt.transcript;
   416→          break;
   417→        }
   418→      }
   419→      if (transcript) setInput((prev) => (prev ? `${prev} ${transcript}`.slice(0, MAX_MESSAGE_LENGTH) : transcript));
   420→    };
   421→    rec.onerror = () => {
   422→      setListening(false);
   423→      toast(t("ask.voiceUnavailable"));
   424→    };
   425→    rec.onend = () => setListening(false);
   426→    recognitionRef.current = rec;
   427→    try {
   428→      rec.start();
   429→      setListening(true);
   430→    } catch {
   431→      setListening(false);
   432→      toast(t("ask.voiceUnavailable"));
   433→    }
   434→  }, [listening]);
   435→
   436→  useEffect(() => {
   437→    return () => {
   438→      recognitionRef.current?.stop();
   439→      recognitionRef.current = null;
   440→    };
   441→  }, []);
   442→
   443→  // ------------------------------------------------------------- autoscroll / autosize
   444→
   445→  useEffect(() => {
   446→    const el = inputRef.current;
   447→    if (!el) return;
   448→    el.style.height = "auto";
   449→    el.style.height = `${Math.min(el.scrollHeight, 120)}px`;
   450→  }, [input]);
   451→
   452→  useEffect(() => {
   453→    const el = listRef.current;
   454→    if (!el) return;
   455→    const farFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight > 400;
   456→    el.scrollTo({ top: el.scrollHeight, behavior: farFromBottom ? "auto" : "smooth" });
   457→  }, [messages.length, pending?.question, pending?.status]);
   458→
   459→  // ------------------------------------------------------------- render helpers
   460→
   461→  const meReady = !me.isLoading;
   462→  const noProfile = meReady && !hasProfile;
   463→  const showEmpty = meReady && hasProfile && !pending && !messagesLoading && !messagesError && messages.length === 0;
   464→  const hasAssistantReply = messages.some((m) => m.role === "assistant");
   465→  const canSend = input.trim().length > 0 && pending === null;
   466→
   467→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   468→    if (e.key === "Enter" && !e.shiftKey) {
   469→      e.preventDefault();
   470→      if (canSend) void send(input);
   471→    }
   472→  };
   473→
   474→  return (
   475→    <ScreenScaffold bare>
   476→      {/* -------------------------------------------------- header */}
   477→      <header className="flex shrink-0 items-center justify-between gap-2 px-4 pb-2.5 pt-4">
   478→        <div className="min-w-0">
   479→          <p className="text-[10.5px] font-semibold uppercase tracking-[0.14em] text-muted-foreground">{t("nav.ask")}</p>
   480→          <h1 className="truncate font-display text-[19px] font-semibold leading-tight text-foreground">
   481→            {activeConversation?.title ?? t("ask.title")}
   482→          </h1>
   483→        </div>
   484→        <div className="flex shrink-0 items-center gap-1.5">
   485→          <HeaderIconButton icon={SquarePen} label={t("ask.newChatAction")} onClick={startNew} />
   486→          <HeaderIconButton icon={History} label={t("ask.history")} onClick={() => setHistoryOpen(true)} />
   487→        </div>
   488→      </header>
   489→
   490→      {/* -------------------------------------------------- message list */}
   491→      <div
   492→        ref={listRef}
   493→        role="log"
   494→        aria-live="polite"
   495→        aria-label={t("ask.title")}
   496→        className="scroll-thin min-h-0 flex-1 overflow-y-auto px-4 pb-6"
   497→      >
   498→        {noProfile ? (
   499→          <EmptyState
   500→            icon={MoonStar}
   501→            title={t("ask.needProfileTitle")}
   502→            body={t("ask.needProfile")}
   503→            actionLabel={t("ask.needProfileCta")}
   504→            onAction={() => push({ id: "profile.birthEdit", params: { mode: "create" } })}
   505→          />
   506→        ) : messagesLoading ? (
   507→          <ConversationSkeleton />
   508→        ) : messagesError ? (
   509→          <ErrorState
   510→            title={t("ask.errorTitle")}
   511→            body={errorMessage(messagesQuery.error)}
   512→            onRetry={() => void messagesQuery.refetch()}
   513→          />
   514→        ) : showEmpty ? (
   515→          <EmptyAsk busy={pending !== null} onAsk={(q) => void send(q)} />
   516→        ) : (
   517→          <div className="space-y-4">
   518→            {messages.map((m, i) =>
   519→              m.role === "user" ? (
   520→                <UserBubble key={m.id} content={m.content} />
   521→              ) : (
   522→                <AssistantCard
   523→                  key={m.id}
   524→                  m={m}
   525→                  expanded={!!expanded[m.id]}
   526→                  feedbackValue={feedbackSent[m.id]}
   527→                  disabled={pending !== null}
   528→                  speakState={speakState?.id === m.id ? speakState.status : "idle"}
   529→                  retryQuestion={
   530→                    m.failed && messages[i - 1]?.role === "user" ? messages[i - 1].content : undefined
   531→                  }
   532→                  onToggleFactors={toggleFactors}
   533→                  onAsk={(q) => void send(q)}
   534→                  onCopy={(c) => void copyAnswer(c)}
   535→                  onListen={(id) => void toggleListen(id)}
   536→                  onShare={(c) => void shareAnswer(c)}
   537→                  onFeedback={(id, v) => void sendFeedback(id, v)}
   538→                />
   539→              )
   540→            )}
   541→            {pending ? (
   542→              <PendingView pending={pending} onStop={stopThinking} onRetry={retryPending} />
   543→            ) : null}
   544→          </div>
   545→        )}
   546→      </div>
   547→
   548→      {/* -------------------------------------------------- talk to an astrologer */}
   549→      {hasProfile && hasAssistantReply && !pending ? (
   550→        <button
   551→          type="button"
   552→          onClick={() => openInTab("astrologers", { id: "astrologers.list" })}
   553→          className="press mx-3 mb-1.5 flex shrink-0 items-center gap-3 rounded-2xl border border-dashed border-hairline bg-secondary/50 p-3 text-left hover:bg-secondary"
   554→        >
   555→          <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-card text-muted-foreground">
   556→            <Users className="h-5 w-5" strokeWidth={1.75} />
   557→          </span>
   558→          <span className="min-w-0 flex-1">
   559→            <span className="block truncate text-[13.5px] font-semibold text-foreground">{t("common.talkToAstrologer")}</span>
   560→            <span className="block truncate text-[12px] text-muted-foreground">{t("ask.talkToAstrologerBody")}</span>
   561→          </span>
   562→          <ChevronRight className="h-4 w-4 shrink-0 text-muted-foreground" />
   563→        </button>
   564→      ) : null}
   565→
   566→      {/* -------------------------------------------------- composer */}
   567→      {hasProfile ? (
   568→        <div className="mb-[calc(88px+env(safe-area-inset-bottom))] shrink-0 border-t border-hairline bg-background px-3 pb-2 pt-2.5">
   569→          <div
   570→            className={cn(
   571→              "flex items-end gap-1.5 rounded-3xl border border-hairline bg-card px-1.5 py-1.5",
   572→              listening ? "border-primary/50 ring-2 ring-primary/15" : "focus-within:border-primary/40"
   573→            )}
   574→          >
   575→            <textarea
   576→              ref={inputRef}
   577→              rows={1}

   682→/** Astrological key terms that get quiet emphasis inside readings. */
   683→const TERM_RE = new RegExp(
   684→  [
   685→    "\\b(Sun|Moon|Mars|Mercury|Jupiter|Venus|Saturn|Rahu|Ketu)\\b", // planets
   686→    "\\b([1-9]|1[0-2])(st|nd|rd|th)\\s+(house|House)\\b", // houses
   687→    "\\bmahadasha\\b|\\bantardasha\\b|\\bdasha\\b", // life phases
   688→    "\\bnakshatra\\b|\\bascendant\\b|\\btransit(s)?\\b", // chart vocabulary
   689→  ].join("|"),
   690→  "g"
   691→);

   693→/** Renders reading text with key astrological terms quietly bolded. */
   694→function HighlightedText({ text }: { text: string }) {
   695→  const parts: React.ReactNode[] = [];
   696→  let last = 0;
   697→  for (const match of text.matchAll(TERM_RE)) {
   698→    const index = match.index ?? 0;
   699→    if (index > last) parts.push(text.slice(last, index));
   700→    parts.push(
   701→      <strong key={index} className="font-semibold text-foreground">
   702→        {match[0]}
   703→      </strong>
   704→    );
   705→    last = index + match[0].length;
   706→  }
   707→  if (last < text.length) parts.push(text.slice(last));
   708→  return <>{parts}</>;
   709→}
